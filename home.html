POST /login
Public
Login user - Returns user data and API key
const response = await fetch('http://localhost:8000/login', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
        email: 'john@example.com',
        password: 'securepass123'
    })
});
const data = await response.json();
if (data.success) {
    localStorage.setItem('api_key', data.api_key);
    console.log('Welcome', data.user.name);
}
import requests
response = requests.post('http://localhost:8000/login', json={
    'email': 'john@example.com',
    'password': 'securepass123'
})
data = response.json()
if data['success']:
    api_key = data['api_key']
    print(f"Welcome {data['user']['name']}")
curl -X POST http://localhost:8000/login \
  -H "Content-Type: application/json" \
  -d '{
    "email": "john@example.com",
    "password": "securepass123"
  }'

POST /apikey/gen
Public
Generate a new API key for a user (requires user_id)

    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ user_id: 1 })
});
const data = await response.json();
console.log('API Key:', data.api_key);
import requests
response = requests.post('http://localhost:8000/apikey/gen', json={
    'user_id': 1
})
data = response.json()
print('API Key:', data['api_key'])
curl -X POST http://localhost:8000/apikey/gen \
  -H "Content-Type: application/json" \
  -d '{"user_id": 1}'

    })
});
const data = await response.json();
